Recipes for common components, written with Tailwind CSS. Each one is
conflict-free: it sets each CSS property of an element in one place, so it
works with the default join, without tailwind-merge (see
[Writing conflict-free recipes](/recipe/class-recipe/conflict-free-recipes/)).
Copy one, then change its classes to match your design.

## Button

`state` replaces separate `loading` and `disabled` variants, which would
both set the cursor and the opacity:

<RecipePlayground name="class-recipe/cookbook/button" />

The component derives `state` from its props:

```tsx
import type { ComponentProps } from "react";
import type { VariantsOf } from "@lynstack/class-recipe";

type ButtonProps = ComponentProps<"button"> &
  Omit<VariantsOf<typeof button>, "state"> & { readonly loading?: boolean };

export function Button({
  tone,
  size,
  loading = false,
  disabled = false,
  className,
  ...props
}: ButtonProps) {
  const state = disabled ? "disabled" : loading ? "loading" : "idle";
  return (
    <button
      className={button({ tone, size, state, className })}
      disabled={disabled || loading}
      aria-busy={loading}
      {...props}
    />
  );
}
```

## Form field

A label, an input, and a message, whose colors follow `invalid`:

<RecipePlayground name="class-recipe/cookbook/field" />

```tsx
import { useId, type ComponentProps } from "react";
import type { VariantsOf } from "@lynstack/class-recipe";

type FieldProps = Omit<ComponentProps<"input">, "size"> &
  VariantsOf<typeof field> & {
    readonly label: string;
    readonly message?: string;
  };

export function Field({ label, message, size, invalid, ...props }: FieldProps) {
  const id = useId();
  const classes = field({ size, invalid });
  return (
    <div className={classes.root}>
      <label htmlFor={id} className={classes.label}>
        {label}
      </label>
      <input
        id={id}
        className={classes.input}
        aria-invalid={invalid}
        aria-describedby={message === undefined ? undefined : `${id}-message`}
        {...props}
      />
      {message !== undefined && (
        <p id={`${id}-message`} className={classes.message}>
          {message}
        </p>
      )}
    </div>
  );
}
```

`size` is also an attribute of `<input>`, so the props omit it (see
[Building components](/recipe/class-recipe/building-components/#variant-names-that-html-also-uses)).
