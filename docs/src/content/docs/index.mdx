import { Card, CardGrid, LinkCard } from "@astrojs/starlight/components";
import { speedup } from "../../measurements/class-recipe.ts";

<CardGrid>
  <Card title="Fast" icon="rocket">
    A recipe compiles its config once and caches each selection, so most calls
    are a lookup: about {speedup} times as many calls per second as
    class-variance-authority.
  </Card>
  <Card title="Type-safe" icon="approve-check">
    Variants, options, and slots are inferred from the config. An unknown option
    is a type error, and a variant without a default is required.
  </Card>
  <Card title="Pluggable" icon="puzzle">
    Bring your own join function, such as `twMerge`. It runs once per selection,
    not on every call.
  </Card>
  <Card title="Any value" icon="setting">
    Recipes are not tied to class names: define a kind of recipe for style
    objects, or any value you can combine.
  </Card>
</CardGrid>
